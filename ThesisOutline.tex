\phantomsection\section*{Introduction}\label{sec:intro}

\qquad
In this expository thesis, we start Part I by introducing some foundational concepts from continuous time martingale theory.
We then make use of some analysis tools to define a concept essential to the general theory of optimal stopping, namely the essential supremum.
Part I will then finish with motivation and definition of the Markov property.
In Part II, we present the general theory of optimal stopping in non-Markovian models and find a solution in terms of the Snell Envelope.
We finish Part II with motivation for what we hope the Markov property will give us.
In Part III, we look at a specific problem, namely that of valuing the perpetual American put option.
The perpetual American put option also serves as a great example of the power of Markovian structure.
In Part IV, we use the machinery and intuition we have built up to address an Ornstein-Uhlenbeck optimal stopping problem from [PS].

\phantomsection\subsection*{\quad I.I Martingale Theory}\label{sec:I.I}
The definitions and theorems in this section are primarily motivated by the first three chapters of [KS] and the first four chapters of [RY].

\begin{defn} Let $\cFPS$ be a continuous time filtered probability space and consider an $\cFi$-adapted stochastic process $(X_t)_{t \geq 0}$.
We say that $X_t$ is a \textbf{submartingale} with respect to $\cFi$ if \\
\[
    \forall t < s \spc \spc X_t \leq \E[ X_s \quantifyspace \cF_t] \spc  \als
\]
and $X_t$ is $\cF_t$-measurable and is integrable w.r.t. $\Prb$ for all $t \geq 0$. \\
If $X_t $ is $\cF_t$-measurable and $X_t$ is integrable for all $t \geq 0$, and the reverse inequality holds, we say that $(X_t)_{t \geq 0}$ is a \textbf{supermartingale} with respect to $\cFi$. \\
If $(X_t)_{t \geq 0}$ is both a submartingale and a supermartingale with respect to $\cFi$ then we say that $(X_t)_{t \geq 0}$ is a \textbf{martingale} with respect to $\cFi$. \\
\end{defn}

\begin{defn}
    Two stochastic processes $(X_t)_{t \geq 0}$ and $(Y_t)_{t \geq 0}$ on $\PS$ are said to be \textbf{modifications} of each other if
    \[
        \Prb[X_t = Y_t] = 1 \spc \forall t \geq 0.
    \]
    $X$ and $Y$ are further said to be \textbf{indistinguishable} if
    \[
        \Prb[X_t = Y_t \sspc \forall t \geq 0] = 1.
    \]
\end{defn}

We now build some machinery that will be needed to prove the Upcrossing Inequality.

\begin{defn}
    Given $(X_t)_{t \geq 0}$ a stochastic process on $\PS$, for a finite set of times $F \subset [s,t] \subset [0,\infty)$ and real-valued constants $a < b$,
    we define $T_1 := \min \{u \in F \quantifyspace X_u < a \}$ and $T_2 := \min \{u \in F \quantifyspace u > T_1, X_u > b\}$. \\
    Now we define inductively $T_{2k - 1} := \min \{u \in F \quantifyspace u > T_{2k - 2}, X_u < a\}$ and $T_{2k} := \min \{u \in F \quantifyspace  u > T_{2k-1}, X_u > b\}$. \\
    We then define $U_F (X, [a,b]) := \max \{k \quantifyspace T_{2k} \leq t\}$. \\
    And for any infinite set of times $I$ we define,
    \[
        U_I(X, [a,b]) := \sup_{F \subset I, F \text{ finite}} U_F(X, [a,b]).
    \]
\end{defn}

\begin{lemma} \textbf{Upcrossing Inequality} \\
    Given $(X_t)_{t \geq 0}$ a right-continuous submartingale and times $s < t$, we have,
    \[
        \E[U_{[s,t]}(X, [a,b])] \leq \Big ( \E[X_t^+] + |a| \Big ) (b - a)^{-1}.
    \]
    This implies that $\Prb [U_{[s,t]}(X, [a,b]) = \infty] = 0$. \\

    \proof
        We first prove that the result holds for $(M_n)_{n \geq 0}$ a discrete time submartingale over finitely many time steps.  \\
        It will suffice to show that for fixed $N \in \N$, we have $\E [U_{\{0,1,..,N\}}(X, [a,b])] \leq \Big ( \E[M_N^+] + |a| \Big ) (b - a)^{-1}$. \\
        Now take $H_n := \sum_{j=0}^\infty \ind{T_{2j + 1} \leq n \leq T_{2j + 2}}$. \\
        And take $N_n := a \vee M_n = a + (M_n - a)^+$; since $a \vee \cdot$ is convex and non-decreasing, we can apply conditional Jensen's and see that for $n \leq m$,
        \[
            N_n = a \vee M_n \leq a \vee \E[M_m \quantifyspace \cF_n] \leq \E[N_m \quantifyspace \cF_n].
        \]
        And so $N_n$ is a submartingale and, since $1 - H_n \geq 0$, we have that $((1 - H) \cdot N)_n$ is a submartingale. \\
        Also, by construction, $(H \cdot N)_n \geq (b - a) U_{\{0, 1, .., n\}}([a,b])$.
        % \textit{The above can be heuristically interpreted as a portfolio strategy of buying $N$ above $b$ and selling below $a$ is a submartingale, since for every passage from above $b$ to below $a$ we lose at least $b-a$ we can hope to bound the expected number of times this happens as each of them incurs loss} \\
        We find that for fixed $n \in \N$,
        \[
            0 = ((1 - H) \cdot N)_0 \leq \E[(1 \cdot N)_n - (H \cdot N)_n],
        \]
        and so,
        \[
            (b - a) \E[U_{\{0,1..,n\}}(X, [a,b])] \leq \E[(H \cdot N)_n] \leq \E[N_n - N_0] =
        \]
        \[
            \E[(M_n - a)^+ - (M_0 - a)^+]
            \leq \E[(M_n - a)^+]
        \]
        \[
            \leq \E[M_n^+] + |a|.
        \]
        This proves the discrete case. \\
        \\
        By choosing $\{q_n\}_{n = 1}^\infty$ to enumerate $\{b\} \cup (\Q \cap [s, t])$, we can take $F_n := \{q_1, .., q_n\}$, and by right continuity of $X_t$ we will get,
        \[
            U_{[s,t]}(X, [a,b]) = \lim_{n \to \infty} U_{F_n}(X, [a,b]).
        \]
        Further restricting to finite times, we find that $\E[U_{F_n}(X, [a,b])] \leq \E[X_t^+] + |a|$ from the discrete case, and since this holds for all $F_n$, by Fatou's lemma we find,
        \[
            \E \big [U_{[s,t]}(X, [a,b]) \big ] = \E \Big [ \liminf_{n \to \infty } U_{F_n}(X, [a,b]) \Big ] \leq
        \]
        \[
            \liminf_{n \to \infty} \E \big [ U_{F_n}(X, [a,b]) \big ] \leq \frac{\E[X_t^+] + |a|}{(b-a)}.
        \]
        \QED

\end{lemma}

\begin{defn}
    We are given a filtered probability space $\cFPS$. 
    An $\cF$-measurable random variable $T: \Omega \to [0,\infty]$ is called a random time. \\
    A random time $T$ is said to be an \textbf{optional time} if $\{T < t\} \in \cF_t$ for all $t \geq 0$.  \\
    A random time $S$ is said to be a \textbf{stopping time} if $\{S \leq t\} \in \cF_t$ for all $t \geq 0$.  \\
    For an optional time $T$ we define the sigma algebras
    \[
        \cF_T := \{A \in \cF \quantifyspace A \cap \{T \leq t\} \in \cF_t \spc \forall t \},
    \]
    \[
        \cF_{T^+} := \{A \in \cF \quantifyspace A \cap \{T < t\} \in \cF_t \spc \forall t \}.
    \]
    Given a process $(X_t)_{t \geq 0}$ and a stopping time $\tau$, we introduce the notation of the stopped process $(X^\tau_t)_{t \geq 0}$; we define $X^\tau_t := X_{t \wedge \tau}$.
\end{defn}

\begin{defn}
    A collection of random variables $(V_\alpha)_{\alpha \in A}$ is said to be \textbf{Uniformly Integrable (U.I.)} if,
    \[
        \lim_{K \to \infty} \sup_{\alpha \in A} \sspc \E [V_\alpha \ind{|V_\alpha| \geq K}] = 0.
    \]
\end{defn}

\begin{thm}
    \textbf{Optional Sampling} $\text{[KS]}$
    Assume that $(M_t)_{t \geq 0}$ is a right-continuous submartingale on $\cFPS$ and that $S, T$ are optional times s.t. $0 \leq S \leq T$. \\
    If $(M_{t \wedge T})_{t \geq 0}$ is U.I. then
    \[
        M_S \leq \E[M_T \quantifyspace \cF_{S^+}].
    \]
    If $S$ is a stopping time then $\cF_S$ can replace $\cF_{S^+}$ above.
\end{thm}

\begin{thm}

\begin{thm}
    \textbf{Doob's Maximal Inequality} \\
    Given $(M_t)_{t \geq 0}$ a submartingale, we define $M_t^* := \sup_{0 \leq s \leq t} M_s^+$ and then for $p \in (1,\infty)$ we have,
    \[
        \E[(M_t^*)^p] \leq \Big ( \frac{p}{p-1} \Big )^p  \sspc \E[(M_t^+)^p].
    \]
\end{thm}

\begin{defn}
    We say that a filtration $\cFi$ for a probability space $\PS$ satisfies the \textbf{usual conditions} if $\mathcal{W} := \{A \in \cF \quantifyspace \Prb[A] \in \{0,1\}\}$ is contained in $\cF_0$
    and $(\cF_t)_{t \geq 0}$ is right continuous, i.e.
    \[
        \forall t \geq 0 \spc \spc \cF_t = \cF_{t^+} = \bigcap_{s > t} \cF_s.
    \]
    Note that under the usual conditions all optional times are stopping times and for a stopping time $\tau$ we have $\cF_\tau = \cF_{\tau^+}$.
\end{defn}

\begin{defn}
    Given $(M_t)_{t \geq 0}$ RC and adapted on $\cFPS$, we say that $(M_t)_{t \geq 0}$ is a \textbf{local martingale}
    if there exists a sequence of stopping times $\{T_n\}_{n =1}^\infty$ s.t. $T_n \nearrow \infty$ a.s.
    and for all $n \in \N$ we have that $(M_{t \wedge T_n})_{t \geq 0}$ is a martingale. \\
    In this case we refer to such a sequence of stopping times as a \textbf{localizing sequence}.
\end{defn}

\begin{lemma} $\text{[PR]}$ % Theorem 51 in 1. Preliminaries
    Say that a local martingale $(M_t)_{t \geq 0}$ is uniformly bounded, i.e., $|M_t| \leq A$ a.s. for all $t$ and $\E[|A|] < \infty$,
    then $(M_t)_{t \geq 0}$ is a true martingale.
\end{lemma}

\begin{defn}
    A stochastic process $(X_t)_{t \geq 0}$ is said to be a \textbf{semimartingale} if there exists a local martingale $(M_t)_{t \geq 0}$ starting at 0 and a finite variation adapted process $(A_t)_{t \geq 0}$ starting at 0 s.t.
    \[
        X_t = X_0 + M_t + A_t.
    \]
\end{defn}

\begin{lemma} $\text{[RY]}$ % Prop 1.2 of ch IV
    A continuous local martingale $(M_t)_{t \geq 0}$ that is of finite variation is indistinguishable from 0. \\
\end{lemma}

\begin{cor}
    Given a continuous semimartingale $X_t$, the decomposition
    \[
        X_t = X_0 + M_t + A_t
    \]
    where $M_t$ is a continuous local martingale starting at 0 and $A_t$ is a continuous finite variation adapted process starting at zero, is unique up to indistinguishability.

    \proof 
    Say $X_t$ is a semimartingale and
    \[
        X_t = X_0 + M_t + A_t = X_0 + M_t' + A_t'
    \]
    with $M_t, M_t'$ being continuous local martingales starting at zero and $A_t, A_t'$ are continuous finite variation processes starting at zero.
    We find,
    \[
        M_t - M_t' = A_t' - A_t
    \]
    And so $M_t - M_t'$ is a continuous local martingale of finite variation, and therefore must be indistinguishable from 0.
    This shows that $M_t$ and $M_t'$ are indistinguishable and in turn that $A_t$ and $A_t'$ are indistinguishable. \\
    \QED 
\end{cor}

\begin{thm} $\text{[RY]}$
    If $M,N$ are continuous path local martingales then there exists a unique process $\la M, N \ra_t$ that starts at 0 and s.t. $(M_t N_t - \la M, N \ra_t)_{t \geq 0}$ is a local martingale.
    Also $\la M, N \ra$ is continuous and has bounded variation paths.
    We refer to $\la M, N \ra$ as the \textbf{cross variation} of $M$ against $N$. \\
    We refer to $\la M, M \ra$ as the \textbf{quadratic variation} of $M$. \\
    In the case of quadratic variation we have that the map $t \mapsto \la M, M \ra_t$ is non-decreasing. \\
\end{thm}

\begin{defn}
    Given continuous semimartingales with decompositions $X_t = X_0 + M_t + A_t$ and $Y_t = Y_0 + M_t' + A_t'$
    where $M_t, M_t'$ are continuous local martingales starting at zero and $A_t, A_t'$ are finite variation adapted processes starting at zero,
    we define the \textbf{cross variation} of $X$ against $Y$, $\la X, Y \ra$ as follows,
    \[
        \la X, Y \ra := \la M, M' \ra.
    \]
\end{defn}

\begin{defn}
    A stochastic process $(B_t)_{t \geq 0}$ starting at 0 and with continuous paths on an event of probability 1 is said to be a \textbf{1-dimensional Brownian motion (Brownian motion/Wiener process)} if:
    \begin{enumerate}
        \item[(i)] For all $s < t$, we have that $(B_t - B_s)$ has a normal distribution with mean $0$ and $\sigma^2 = t-s$.
        \item[(ii)] For all finite collections of deterministic times $0 \leq t_0 < t_1 < t_2 < t_3 < .. < t_n$ we have that the collection of increments as random variables $\{B_{t_k} - B_{t_{k-1}}\}_{k=1}^n$ is independent.
    \end{enumerate}
\end{defn}

\begin{defn}
    A stochastic process $(X_t)_{t \geq 0}$ is \textbf{progressively measurable} with respect to the filtration $\cFi$ if for all $t \geq 0$

    then we say that $(X_t)_{t \geq 0}$ is \textbf{predictable}.
\end{defn}

    Adapted, continuous path processes are predictable.

\begin{defn}
    In a given probability space, we take $\bH^2$ to be the set of right-continuous martingales $(M_t)_{t \geq 0}$ s.t.
    \[
        \sup_t \E[M_t^2] < \infty,
    \]
    in other words, the set of right-continuous martingales that are bounded in $L^2$. \\
    We endow $\bH^2$ with $\norm{M}{\bH} := \big ( \lim_{t \to \infty} \E[M_t^2] \big )^{1/2}$. \\
    We define $H^2$ to be the subset of processes in $\bH^2$ with continuous paths. \\
    We take $H^2_0$ to be those elements of $H^2$ that deterministically assume the value 0 at time 0. \\
\end{defn}

\begin{defn}
    Given some $(M_t)_{t \geq 0} \in H^2$, we let $\cL^2(M)$ denote the set of progressively measurable processes $K$ s.t.
    \[
        \norm{K}{M}^2 := \E \Big [
            \int_0^\infty K_s^2 \df{\la M, M \ra_s}
        \Big ] < \infty.
    \]
    We would like to clarify that the integral inside the expectation is being taken w.r.t. the Lebesgue-Stieltjes measure induced by $t \mapsto \la M, M \ra_t$. \\
    Notice that $\norm{\cdot}{M}$ is a norm and we define $L^2(M)$ to be the equivalence classes of $\cL^2(M)$ under this norm. \\
\end{defn}

\begin{defn}
    For $M$ a continuous local martingale we define $L^2_{loc}(M)$ to be the space of progressively measurable processes $K$ s.t.
    there exists a sequence of stopping times $\{\tau_n\}_{n=1}^\infty$ s.t. $\tau_n \nearrow \infty$ a.s. and
    \[
        \E \Big [
            \int_0^{\tau_n} K_u \df{\la M, M \ra_u}
        \Big ] < \infty \spc \spc \forall n \in \N.
    \]
\end{defn}

\begin{thm} $\text{[RY]}$ % 2.2, 2.7 of Ch 4.2 Stochastic Integration : Stochastic Integrals
    Let $M \in H^2$; for each $K \in L^2(M)$, there is a unique element of $H_0^2$ denoted by $K \cdot M$ s.t. 
    \[
        \la K \cdot M, N \ra = K \cdot \la M, N \ra,
    \]
    for every $N \in H^2$. \\
    If we relax to just having that $M$ is a continuous local martingale and $K \in L^2_{loc}(M)$, then we have that there is a unique local martingale $(K \cdot M)$ starting at 0 satisfying the above for every $N$ that is a continuous local martingale.
\end{thm}

\begin{defn}
    We refer to $K \cdot M$ from the above theorem as the \textbf{stochastic integral} of $K$ with respect to $M$.
    We also use the notation,
    \[
        \int_0^t K_u \df{M_u} := (K \cdot M)_t.
    \]
    Also, given a semimartingale $X = X_0 + M + A$, where $M$ is a continuous local martingale and $A$ is a continuous finite variation adapted process and $K \in L^2_{loc}(M)$, we define the \textbf{stochastic integral} of $K$ with respect to $X$. We again use the notation,
    \[
        \int_0^t K_u \df{X_u} := (K \cdot X)_t := (K \cdot M)_t + (K \cdot A)_t.
    \]
\end{defn}

\begin{lemma}
    \textbf{Itô's Lemma} \\
    Given $(X_t)_{t \geq 0}$, a continuous semimartingale and $f \in C^2(\R)$, we have,
    \[
        f(X_t) = f(X_0) + \int_0^t f'(X_u) \df{X_u} + \frac{1}{2} \int_0^t f''(X_u) \df{\la X, X \ra_u} \spc
        \forall t \in \R_+ \spc \spc \Prb\text{-}\als.
    \]
    This extends to $f$ having a multidimensional input. \\
    Say that $(X_t)_{t \geq 0}$ is a $d$-dimensional semimartingale, meaning $X^k$ is a semimartingale for every dimension $k=1, .., d$. \\
    In the case that we have $f: \R^d \to \R$ in $C^2$ then,
    \[
    f(X_t) = f(X_0) + \sum_{i=1}^d \int_0^t f_{e_i}(X_u) \df{X_u^i} + \frac{1}{2} \sum_{1 \leq j, k \leq d} \int_0^t f_{e_j, e_k}(X_u) \df{\la X^j, X^k \ra_u},
    \]
    where $f_{e_k}$ represents the partial derivative of $f$ in the $e_k$ direction, with $e_k$ being the $k^{th}$ standard basis vector for $\R^d$.
\end{lemma}

\begin{lemma}
    \textbf{Itô-Tanaka-Meyer} \\
    Given $(X_t)_{t \geq 0}$, a continuous semimartingale and $f \in C^1(\R)$
    with an absolutely continuous derivative, we have,
    \[
        f(X_t) = f(X_0) + \int_0^t f'(X_u) \df{X_u} + \frac{1}{2} \int_0^t f''(X_u) \df{\la X, X \ra_u} \spc
        \forall t \in \R_+ \spc \spc \Prb\text{-}\als.
    \]
\end{lemma}


\phantomsection\subsection*{\quad I.II Essential Supremum}\label{sec:I.II}

In this section, we present the main definitions and results of Appendix A from [MMF]. \\

\begin{defn}
Consider a probability space $\PS$ and $\mathcal{Q}$ a nonempty collection of non-negative random variables defined on $\PS$.
The \textbf{essential supremum} of $\mathcal{Q}$ (denoted ess sup $\mathcal{Q}$) is a random variable $X^*$ with the following two properties:

\begin{enumerate}
    \item[(i)] $\forall X \in \mathcal{Q}, \spc X \leq X^* \spc \als$,
    \item[(ii)] $(X \leq Y \sspc \als \sspc \forall X \in \mathcal{Q}) \Rightarrow (X^* \leq Y \sspc \als)$.
\end{enumerate}
\end{defn}

\begin{defn}
    Consider $\mathcal{Q}$ a nonempty collection of non-negative random variables defined on $\PS$.
    Let $A \in \cF$ and $k \in \N$, where $A_1, A_2, .., A_k \in \cF$ comprise a partition of $A$. And let $X_1, X_2, .., X_k \in \mathcal{Q}$.
    We then say that $\pi = (k, A_1, A_2, .., A_k, X_1, X_2, .., X_k)$ is a \textbf{$\mathcal{Q}$-partition of $A$}. \\
    We then let $P_A$ denote the set of all $\mathcal{Q}$-partitions of $A$. \\
\end{defn}

Now we construct $\mu_\pi^\infty: \cF \to \R_+$ as follows,
\[
    A \xmapsto{\mu_\pi^\infty} \E \bigg [ \sum_{j=1}^k X_j 1_{A_j} \bigg ].
\]
We then define $\mu^\infty: \cF \to \R_+$ as,
\[
    A \xmapsto{\mu^\infty} \sup_{\pi \in P_A} \mu_\pi^\infty(A).
\]

\begin{lemma} $\text{[MMF]}$
    The map $\mu^\infty: \cF \to \R_+$ is a measure on $\Omega$ and $\mu^\infty$ is absolutely continuous with respect to $\Prb$.
\end{lemma}

\vspc

\begin{thm} $\text{[MMF]}$
    Consider $\mathcal{Q}$ a collection of random variables defined on the probability space $\PS$.
    Then ess sup $\mathcal{Q}$ exists and is unique up to $\Prb$-a.s. equivalence.

    \proof
    First we show that if the essential supremum of $\mathcal{Q}$ exists, then it is unique up to $\Prb$-a.s. equivalence. \\
    This follows trivially from the second property of the essential supremum. \\
    Say that $A$ and $B$ both satisfy the definition of essential supremum of $\mathcal{Q}$. \\
    Since $X \leq A$ a.s. and $X \leq B$ a.s. for all $X \in \mathcal{Q}$, we then have that $B \leq A$ a.s. and $A \leq B$ a.s., thus $A = B$ a.s.. \\
    \\
    Now we show existence. \\
    Recall that $\mu^\infty << \Prb$ and now define $X^* := \frac{\df{\mu^\infty}}{\df{\Prb}}$, i.e., $X^*$ is the Radon-Nikodym derivative of $\mu^\infty$ w.r.t. $\Prb$. \\
    We claim that $X^*$ satisfies the criterion of essential supremum of $\mathcal{Q}$. \\
    \\
    Fix some $X \in \mathcal{Q}$. \\
    Now fix a test event $A \in \cF$ and consider the $\mathcal{Q}$-partition of $A$, $\pi^* := (1, A, X)$ and observe,
    \[
        \E \big [X 1_A \big ] = \mu_{\pi^*}^\infty(A) \leq \sup_{\pi} \mu_\pi^\infty(A) = \mu^\infty(A) = \E \Big [\frac{\df{\mu^\infty}}{\df{\Prb}} 1_A \Big ] = \E \big [X^* 1_A \big ].
    \]
    Since this holds for all $A \in \cF$, we have that $X \leq X^*$ a.s., and we have shown the first property. \\
    \\
    Now we fix some random variable $Y$ s.t. $X \leq Y$ a.s. for all $X \in \mathcal{Q}$. \\
    Observe,
    \[
        \E \big [X^* 1_A \big ] =
        \sup_{\pi = (k, A_1 .. A_k , X_1 .. X_k) \in P_A } \mu_\pi^\infty (A) =
        \sup_{\pi \in P_A} \E \bigg [\sum_{j=1}^k X_j 1_{A_j} \bigg ]  \leq
    \]
    \[
        \sup_{\pi \in P_A} \E \bigg [ Y \sum_{j=1}^k 1_{A_j} \bigg ] = \E \big [ Y 1_{A} \big ].
    \]
    This shows the second condition. \\
    \QED
\end{thm}

\begin{lemma} $\text{[MMF]}$
    Consider $\mathcal{Q}$ a collection of random variables defined on the probability space $\PS$ s.t. $\mathcal{Q}$ is closed under pairwise maximum, i.e.
    $A,B \in \mathcal{Q}$ implies $A \vee B \in \mathcal{Q}$, then there is a non-decreasing sequence $\{C_n\}_{n=1}^\infty \subset \mathcal{Q}$ s.t. $X^* = \lim_{n \to \infty} C_n$ a.s..
\end{lemma}

\begin{lemma}
    Given a nonempty family of non-negative random variables $\mathcal{Q}$ and a non-negative random variable $J$, we have,
    \[
        J \big ( \esssup \sspc \mathcal{Q} \big ) = \esssup \sspc \big \{J Q \quantifyspace Q \in \mathcal{Q} \big \} \spc \spc \als.
    \]
    \proof
    
    We take $X := \esssup \mathcal{Q}$ and we now want to show that $J X$ satisfies the properties of being the essential supremum of $\{JQ \quantifyspace Q \in \mathcal{Q}\}$. \\
    First take some element $Q \in \mathcal{Q}$. \\
    From the first property of $X$ being essential supremum of $\mathcal{Q}$,
    \[
        Q \leq X \spc \als,
    \]
    and so,
    \[
        JQ \leq J X \spc \als.
    \]
    This shows that $JX$ satisfies the first property. \\
    Now fix some random variable $Z$ s.t. $Z \geq JQ$ a.s. for all $Q \in \mathcal{Q}$. \\
    We now fix a specific $Q \in \mathcal{Q}$ and see,
    \[
        \ind{J > 0} \frac{1}{J}Z \geq \ind{J > 0} Q \spc \als,
    \]
    and now,
    \[
        \ind{J > 0} \frac{1}{J} Z + \ind{J = 0} X \geq Q \spc \als.
    \]
    Since this holds for all $Q$, from the second property of $X$ we get,
    \[
        \ind{J > 0} \frac{1}{J} Z + \ind{J = 0} X \geq X \spc \als,
    \]
    \[
        Z \geq \ind{J > 0} Z \geq J X \spc \als.
    \]
    This shows the second property. \\
    \QED
\end{lemma}

\phantomsection\subsection*{ \quad I.III Markov Property, Strong Markov Property \& Ornstein-Uhlenbeck Processes}\label{sec:I.III}
\cheekynewline
We introduce the idea of the Markov Property, which is motivated by the idea that for some processes the increments are dependent only on the state at the start of the time interval of the increment and not on the entire trajectory. \\
\begin{defn}
    Given a state space $(E, \mathcal{E})$, a \textbf{transition function} is a function
    $P$ defined on the subset of $[0,\infty) \times E \times [0,\infty) \times \mathcal{E}$ with the first argument less than or equal to the third argument and assuming values in $[0,1]$ s.t.
    for all $s,t \in [0,\infty), A \in \mathcal{E}$ we have that $P(s, \cdot; t, A)$ is $\mathcal{E}$-measurable and
    for each $s,t \in [0,\infty), e \in E$ we have that $P(s, e; t, \cdot)$ is a probability measure, and for every $0 \leq s < r < t$ and $e \in E$ we have,
    \begin{equation}
        P(s, e; t, A) = \int_E P(s,e;r, dy) P(r, y; t, A).
        \tag{\textsc{Chapman-Kolmogorov equation}}
    \end{equation}
    If, in addition, for all $s,t \geq 0$ and $\alpha > 0$ we have: \\
    $$P(s, \spdot; s+\alpha, \spdot) = P(t, \spdot; t + \alpha, \spdot)$$ \\
    then we say that $P$ is \textbf{time-homogeneous} and define: \\
    $$P(\alpha, \spdot; \spdot) := P(s, \spdot; s+\alpha, \spdot) = P(t, \spdot; t+\alpha, \spdot).$$
\end{defn}

\begin{defn}
    Given $(X_t)_{t \geq 0}$ as $(E, \mathcal{E})$-valued random elements on the filtered probability space $\cFPS$ s.t. $X_t \in \cF_t$ for all $t \geq 0$
    and given a transition function $P$
    we say that $X_t$ is a \textbf{Markov Process} with respect to $\cFi$ with transition function $P$ if $X_t \in \cF_t$ for all $t$ and for all $s < t$ and measurable functions $f: E \to \R_+$ we have 
    \begin{equation*}
        \E[f(X_t) \quantifyspace \cF_s] = \int_E P(s, X_s; t, dy) f(y) \spc \spc \Prb-\als.
    \end{equation*}
    If, in addition, $P$ is time-homogeneous, then we say that $X_t$ is a \textbf{time-homogeneous} Markov Process. \\
\end{defn}

\begin{defn}
    Given $(X_t)_{t \geq 0}$ a $(E, \mathcal{E})$-valued time-homogeneous Markov process with respect to $\cFi$ with transition function $P$
    we say that $X_t$ satisfies the \textbf{Strong Markov Property} if for all measurable functions $f: E \to \R_+$ and all stopping times $\tau$ we have,
    \[
        \E[f(X_{\tau+\alpha}) \quantifyspace \cF_\tau] = \int_E P(\alpha, X_\tau; dy) f(y) \spc \spc \Prb\text{-}\als \text{ on } \{\tau < \infty\}.
    \]
    \small{ \spc \textit{We note that some authors relax away the requirement that $X$ must be time-homogeneous in order to have the strong Markov property.}}
\end{defn}

We introduce a piece of machinery specific to strong Markov processes that will be useful later:
\begin{defn}
    Given a real-valued strong Markov process $(X_t)_{t \geq 0}$ with transition function $P$, we define the operator $\bL_X$ acting on functions $f: \R \to \oR$ for which the quantity we are about to introduce is well defined. \\
    %We let $\Prb^x$ be a probability measure giving probability of 1 to the event that $X_0 = x$ for some $x \in \R$. ($X_t$ is still a strong Markov process with the same transition function regardless of choice of $x$.) \\
    We take $\Prb^x$ to be a probability measure that makes $X$ strong Markov with transition function $P$ and s.t. $\Prb^x[X_0 = x] = 1$. \\
    We let $\E^x$ represent integration w.r.t. the probability measure $\Prb^x$. \\
    We now introduce the \textbf{infinitesimal generator} $\bL_X$.
    The intuition behind the infinitesimal generator is that we want to capture the infinitesimal change in expectation of a function of $X_t$ from a given starting state $x$. It is defined as follows,
    \[
        (\bL_X f)(x) = \lim_{t \to 0^+}\frac{\E^x[f(X_t)] - f(x)}{t}.
    \]
\end{defn}


We now introduce a specific instance of a class of Markov Processes.
\begin{defn}
    Given $(X_t)_{t \geq 0}$ an adapted stochastic process on the filtered probability space $\cFPS$,
    we say that $X_t$ is an \textbf{Ornstein-Uhlenbeck process} if $X_t$ satisfies the following SDE:
    \[
        \df{X_t} = \theta (\mu - X_t) \df{t} + \sigma \df{W_t},
    \]
    where $W_t$ is a Wiener process and $\theta, \sigma > 0$ and $\mu \in \R$ are constants. \\
\end{defn}

\phantomsection\subsection*{\quad II.I Formulation of General Problem}\label{sec:II.I}
The formulation and presentation used here draws on aspects from both the presentation in Section 2 of [S1] and the presentation in Appendix D of [MMF]. \\
\\
We are given a filtered probability space $\cFPS$ s.t. the filtration satisfies the usual conditions and we specifically impose $\cF_0 = \mathcal{W}$, i.e., $\cF_0$ is sigma algebra of null sets and probability 1 sets. \\
We are also given a time horizon $T \in (0, \infty]$. \\
We are given $(Y_t)_{t \geq 0}$ a non-negative valued and right continuous stochastic process on our probability space. \\
In the case that $T = \infty$ we consider $\cF_\infty := \sigma ( \cup_{t \geq 0} \cF_t)$ and $Y_\infty := \limsup_{t \to \infty} Y_t$. \\
We now define $\mathcal{S}$ to be the set of all $\cFi$ stopping times. \\
Now we take
\[
    \cS_\tau := \{\theta \in \cS \quantifyspace \theta \geq \tau \sspc \als \}.
\]
\cheekynewline

It is useful for intuition to conceive of $Y_t$ as representing the value that the exercise of a particular derivatives contract (for now just a single exercise derivative) at time $t$ yields.

\begin{obj} $\text{[MMF]}$
    We want to achieve the following:
    \begin{enumerate}
        \item[(i)] Compute the maximal expected value of stopping $Y$
            \[
                Z_0 := \sup_{\tau \in \cS} \E[Y_\tau].
            \]
        \item[(ii)] Find conditions that yield existence of a stopping time $\tau_*$ that achieves the value in the prior sub-objective.
        \item[(iii)] Build understanding of $\tau_*$ and characterize $\tau_*$.
    \end{enumerate}
\end{obj}

\phantomsection\subsection*{\quad II.II The Snell Envelope}\label{sec:II.II}
The essential piece of intuition we need to start to build the right machinery
is that, given two stopping times $\tau, \theta \in \cS$, we can construct another stopping time $\gamma$ which takes values of $\tau$ when preferable and values of $\theta$ when preferable.
Specifically, we define $A := \{\E[Y_\tau - Y_\theta \quantifyspace \cF_{\tau \wedge \theta}] \geq 0\}$ and then define $\gamma := \tau 1_A + \theta 1_{A^c}$.
This observation tells us that
given a fixed $\tau \in \cS$, we have that $\{\E[Y_\theta \quantifyspace \cF_\tau] \}_{\theta \in \cS_\tau}$ is closed under pairwise maximization.
At a glance, one might hope that this family would have a maximal element in an $\omega$ by $\omega$ sense, but we must recall that each of these elements is only defined up to a.s. equality, so if we tried to make sense of such an $\omega$ by $\omega$ supremum we could get the random variable $\omega \mapsto \infty$. 
This should be enough to tell you that the essential supremum is the right tool. We can now get to the main definition. 
\\
\\
In the remainder of this subsection we present the main results of Appendix D from [MMF] about the Snell envelope. \\


\begin{defn}
    We define the following,
    \[
        Z_v := \esssup_{\tau \in \cS_v} \E[Y_\tau \quantifyspace \cF_v].
    \]
    A modification of this will be the "Snell Envelope".
\end{defn}

\begin{assume}
    We only consider the non-trivial case where
    \[
        0 < Z_0 = \sup_{\tau \in \cS} \E[Y_\tau] < \infty,
    \]
    and so we will assume this going forward. \\
\end{assume}

\begin{assume}
    We now impose the condition that
    \[
        \E[\sup_{t \in [0,T]} Y_t] < \infty.
    \]
    Using this condition, we can achieve sub-objective (ii).
\end{assume}
\cheekynewline

\begin{lemma} $\text{[MMF]}$
    For all $v \in \cS$ and $\tau \in \cS_v$, the family $\{\E[Y_\rho \quantifyspace \cF_v]\}_{\rho \in \cS_v}$ is closed
    under pairwise maximization. Furthermore, there is a sequence $\{\rho_n\}_{n=1}^\infty$ of stopping times in $\cS_\tau$ s.t.
    the sequence $\{\E [Y_{\rho_n} \quantifyspace \cF_v] \}_{n=1}^\infty$ is non-decreasing and
    \[
        \esssup_{\rho \in \cS_\tau} \E[Y_\rho \quantifyspace \cF_v] = \lim_{n\to \infty} \E[Y_{\rho_n} \quantifyspace \cF_v].
    \]
    \proof
    The first statement of this lemma is exactly our observation at the beginning of this Snell Envelope section, and the second statement is a direct result of the first statement and Lemma 9 in the Essential Supremum section. \QED \\
\end{lemma}

The reader may note that $Z_v$ is a random variable that has been defined as a function of a stopping time $v \in \cS$. It is not clear that this definition of $Z$ is consistent; that is, if we consider the process given by $Z$ at deterministic times, we do not know that stopping this process at $v$ is the same as what we defined as $Z_v$. We address this issue now and build some understanding of the relation between $Z$ and $Y$.
\begin{claim} $\text{[MMF]}$
    For all $v, \sigma \in \cS$ and $\tau \in \cS_v$, we have
    \[
        Z_v = Z_\sigma \spc \als \text{ on } \{\sigma = v\},
    \]
    \[
        \E[Z_\tau \quantifyspace \cF_v] = \esssup_{\rho \in \cS_\tau} \E[Y_\rho \quantifyspace \cF_v] \spc \als,
    \]
    \[
        \E[Z_\tau \quantifyspace \cF_v] \leq Z_v \spc \als,
    \]
    \[
        \E[Z_\tau] = \sup_{\rho \in \cS_\tau} \E[Y_\rho] \leq Z_0 < \infty.
    \]

    \proof 
    
    We start by showing the first statement. \\
    Apply Lemma 10,
    \[
        Z_v \ind{\sigma = v} = \esssup_{\rho \in \cS_v} \E[Y_\rho \ind{\sigma = v} \quantifyspace \cF_v] \spc \als.
    \]
    We make a sub-claim about conditional expectation and stopping times, namely, for an $\cF$-measurable random variable $X$ and stopping times $a, b$ we have,
    \[
        \E[X \quantifyspace \cF_a] \ind{a = b} = \E[X \quantifyspace \cF_b] \ind{a = b} \spc \als.
    \]
    Proof of sub-claim:
    Note that $\ind{a = b}$ is in both $\cF_a$ and $\cF_b$. \\
    We will try to show that the right hand side satisfies the properties of the conditional expectation $\E[X \ind{a = b} \quantifyspace \cF_a]$. \\
    We need to show that the right hand side is $\cF_a$ measurable and yields the proper value when integrated over $\cF_a$ events. \\
    Fix some $O \in \cB(\R)$. \\
    Take $\gamma := \E[X \quantifyspace \cF_b]$, by definition $\gamma$ is $\cF_b$ measurable.
    Now, in the case that $0 \not \in O$,
    \[
        (\gamma \ind{a = b})^{-1}(O) = \{a = b\} \cap \gamma^{-1}(O).
    \]
    We define $S := \gamma^{-1}(O) \in \cF_b$. \\
    We want to show that $\{a = b\} \cap S \in \cF_a$. In fact, we show this for general $S \in \cF_b$. \\
    Fix $t \geq 0$. \\
    Note that since $S \in \cF_b$ we have,
    \[
        S \cap \{b \leq t\} \in \cF_t.
    \]
    And now,
    \[
        S \cap \{a = b\} \cap \{a \leq t\} =
        \{a = b \leq t\} \cap \big ( S \cap \{b \leq t\} \big ) \in \cF_t.
    \]
    This shows that $S \cap \{a = b\}$ is in $\cF_a$. \\
    The other case is that $0 \in O$ and, in this case, $(\gamma \ind{a = b})^{-1}(O) = \gamma^{-1}(O) \cup \{a \neq b\} = \big (\gamma^{-1}(O^c) \cap \{a = b\} \big )^c$, but we know $\gamma^{-1}(O^c) \in \cF_b$ and so $\gamma^{-1}(O^c) \cap \{a = b\} \in \cF_a$ and hence its complement is also in $\cF_a$. \\
    We now see that $\E[X \quantifyspace \cF_b] \ind{a=b} \in \cF_a$ and, reversing these roles, we see $\E[X \quantifyspace \cF_a] \ind{a = b} \in \cF_b$, thus seeing that both of these random variables are $\cF_{a \wedge b}$-measurable. \\
    And for $A \in \cF_a$ using what we know about measurability $A \cap \{a = b\} \in \cF_b$,
    \[
        \E[X \ind{a = b} 1_A] =
        \E[\E[X 1_{A \cap \{a = b\}} \quantifyspace \cF_b]] =
        \E[1_A \ind{a = b} \E[X \quantifyspace \cF_b]] =
    \]
    \[
        \E[1_A \E[X \ind{a = b} \quantifyspace \cF_b]] \spc \als.
    \]
    End of proof of subclaim. \\
    \\
    And now we fix some $\rho \in \cS_v$ and define $\rho' := \rho \ind{\sigma = v} + T \ind{\sigma \neq v}$:
    \[
        \E[Y_\rho \ind{v = \sigma} \quantifyspace \cF_v] =
        \E[Y_\rho \ind{v = \sigma} \quantifyspace \cF_\sigma] =
        \E[Y_{\rho'} \ind{v = \sigma} \quantifyspace \cF_\sigma] \leq
    \]
    \[
        \Big ( \esssup_{\kappa \in \cS_\sigma} \E[Y_\kappa \quantifyspace \cF_\sigma] \Big ) \ind{v = \sigma} = Z_\sigma \spc \als.
    \]
    Taking the essential supremum over all $\rho$, the left-hand side becomes $Z_v \ind{v = \sigma}$, and we have $Z_v \ind{v = \sigma} \leq Z_\sigma \ind{v = \sigma}$.
    Swapping $v$ and $\sigma$ yields the reverse inequality, and so $Z_v = Z_\sigma$ a.s. on $\{v = \sigma\}$. \\
    \\
    Now for the second statement we invoke Lemma 9 to get $\{\rho_n\}_{n=1}^\infty \subset \cS_\tau$ s.t. $\{\E[Y_{\rho_n} \quantifyspace \cF_\tau]\}_{n=1}^\infty$ is non-decreasing and $Z_\tau = \lim_{n \to \infty} \E[Y_{\rho_n} \quantifyspace \cF_\tau]$. \\
    By monotone convergence theorem for conditional expectation,
    \[
        \E[Z_\tau \quantifyspace \cF_v] = \lim_{n \to \infty} \E[ \E[Y_{\rho_n} \quantifyspace \cF_\tau] \quantifyspace \cF_v] = 
        \lim_{n \to \infty} \E[ Y_{\rho_n} \quantifyspace \cF_v ] \leq
    \]
    \[
        \esssup_{\rho \in \cS_\tau} \E[Y_\rho \quantifyspace \cF_v].
    \]
    And for $\rho \in \cS_\tau$ we have,
    \[
        Z_\tau \geq \E[Y_\rho \quantifyspace \cF_\tau] \spc \text{ and }\spc
        \E[ Z_\tau \quantifyspace \cF_v] \geq \E[Y_\rho \quantifyspace \cF_v].
    \]
    Since the above holds for all $\rho \in \cS_\tau$ we get $\E[Z_\tau \quantifyspace \cF_v] \geq \esssup_{\rho \in \cS_\tau} \E[Y_\rho \quantifyspace \cF_v]$. \\
    This yields the second statement. \\
    \\
    The third statement is a direct consequence of the fact that $\cS_\tau \subset \cS_v$. \\
    The fourth statement follows from Lemma 9 and the third statement where $v$ is selected to be 0. \\
    \QED
\end{claim}

\begin{defn}
    For a given $v \in \cS$ we introduce $\cS_v^* := \{\tau \in \cS_v \quantifyspace \tau > v \sspc \als \text{ on } \{v < T\} \}$ and we define,
    \[
        Z^*_v := \esssup_{\tau \in \cS_v^*} \E[Y_\tau \quantifyspace \cF_v].
    \]
\end{defn}
Observe that our proof of Claim 1 also holds if we replace $Z$ with $Z^*$, so $Z^*$ has all the outlined properties. \\
We now want to show right continuity of a modification of $Z^*$:
\begin{claim} $\text{[MMF]}$
    For any $v \in \cS$ and a sequence $v_n \subset \cS_v^*$ s.t. $\lim_{n \to \infty} v_n = v$ a.s., we have,
    \[
        Z_v = Z_v^* \vee Y_v \spc \als,
    \]
    \[
        \int_A Z_v^* \df{\Prb} = \lim_{n \to \infty} \int_A Z_{v_n}^* \df{\Prb} \spc \spc \forall A \in \cF_v.
    \]
\end{claim}

\begin{cor} $\text{[MMF]}$
    For any $v \in \cS$, we have $Z_v^* = Z_v$ a.s..
\end{cor}

\begin{defn}
    Since $Z$ (as a function from stopping times to random variables) is consistent in the sense of Claim 1, we take the deterministic times and consider the supermartingale $(Z_t)_{t \geq 0}$.
    Since $t \mapsto \E [Z_t^*] = \E[Z_t]$ is right-continuous, we may apply Theorem 2 to see that there exists a supermartingale $(Z^0_t)_{t \geq 0}$ with RCLL paths that is a modification of $Z$.
    We call $Z^0$ the \textbf{Snell Envelope} of $Y$.
\end{defn}

\begin{thm} $\text{[MMF]}$
    Any stopping time $\tau_*$ is optimal starting from time 0 (meaning $\E[Y_{\tau_*}] = Z^0_0 = \sup_{\rho \in \cS} \E[Y_\rho]$) if and only if
    \[
        Z^0_{\tau_*} = Y_{\tau_*} \spc \als,
    \]
    and the stopped supermartingale
    \[
        \big ( Z^0_{t \wedge \tau_*} \big )_{t \geq 0} \text{ is a martingale.}
    \]
    \proof
    First we assume that $\tau_*$ is optimal. \\
    From the definition of $Z$ we have,
    \[
        Z_{\tau_*}^0 \geq Y_{\tau_*} \spc \spc \als.
    \]
    From optimality of $\tau_*$,
    \[
        \E[Y_{\tau_*}] \leq \E[Z_{\tau_*}^0] = Z_0^0 = \E[Y_{\tau_*}],
    \]
    and so,
    \[
        \E[ Z_{\tau_*}^0 - Y_{\tau_*}] = 0.
    \]
    And since the integrand above is $\geq 0$ a.s., we must have that the integrand is equal to 0 a.s. Thus,
    \[
        Z_{\tau_*}^0 = Y_{\tau_*} \spc \spc \als.
    \]
    Now it remains to show that $(Z^0)^{\tau_*}$ is a martingale. \\
    Fix $\theta$ any stopping time less than or equal to $\tau_*$. \\
    From Claim 2 and from the definition of $Z$ along with $Z_{\tau_*}^0 = Y_{\tau_*}$ a.s., we get,
    \[
        \E[Y_{\tau_*} \quantifyspace \cF_\theta] = \E[Z_{\tau_*}^0 \quantifyspace \cF_\theta] \leq Z^0_\theta \leq \E[Y_{\tau_*} \quantifyspace \cF_\theta] \spc \spc \als.
    \]
    And so we find $Z_\theta^0 = \E[Y_{\tau_*} \quantifyspace \cF_\theta]$. \\
    The fact that $(Z^0)^{\tau_*}$ is a martingale then follows directly from the tower property of conditional expectation. \\
    \\
    Now we assume that $\tau_*$ is such that $Z_{\tau_*}^0 = Y_{\tau_*}$ a.s.
    and $(Z^0)^{\tau_*}$ is a martingale. \\
    Using our two assumptions directly we find,
    \[
        \sup_{\rho \in \cS} \E[Y_\rho] = Z_0^0 = \E[Z_{\tau_*}^0 \quantifyspace \cF_0] = \E[Z_{\tau_*}] = \E[Y_{\tau_*}] \spc \spc \als.
    \]
    \QED
\end{thm}

\begin{thm} $\text{[MMF]}$
    Given a stopping time $\tau \in \cS$ we define the following,
    \[
        \theta_\tau^* := \inf \{t \geq \tau \quantifyspace Z^0_t = Y_t \} \wedge T.
    \]
    We claim that $\E[Y_{\theta_\tau^*}] = \sup_{\rho \in \cS_\tau} \E[Y_\rho]$, i.e., $\theta_\tau^*$ is the optimal stopping time beginning from time $\tau$.
\end{thm}

\phantomsection\subsection*{\quad II.III Markov Motivation}\label{sec:II.III}
At a high level, we can think of the (strong) Markov property for a process $(X_t)_{t \geq 0}$ as the property that increments starting from a time $t$ are dependent only on the current state and not on the entire trajectory up to time $t$.
We now consider $(X_t)_{t \geq 0}$ having the strong Markov property and with state space $(\R, \cB(\R))$.
Given our high-level description of the (strong) Markov property, it may seem intuitive that the optimal stopping of $(X_t)_{t \geq 0}$ would depend only on the state at time of stoppage, i.e., that there must be a set of states $S \in \cB(\R)$ s.t. $\tau_* := \inf \{ t \geq 0 \quantifyspace X_t \in S \}$ is optimal;
however, this requires a considerable amount of machinery to prove.
A full treatment of this general task may be seen in [OSR].
Nonetheless, in certain applications
we can use our Markov intuition to guide our search for explicit solutions for the Snell envelope
which we then verify with the help of other tools, namely Itô's lemma.
We will see examples of this in Chapters III and IV. 

\phantomsection\section*{Part III: Perpetual American Put Option}\label{sec:part3}

\phantomsection\subsection*{\quad III.I Black-Scholes Model}\label{sec:III.I}
The presentation of the Black-Scholes Model is inspired by that seen in [S1]. \\
We consider the probability space $\PS$ and $(W_t)_{t \geq 0}$ a Brownian motion on this space. \\
We take $\cFi$ to be the filtration generated by $W$ enlarged by null sets. \\
We take $(X_t)_{t \geq 0}$ to be defined as follows,
\[
    X_t = x \exp \bigg [ \Big ( r - \frac{\sigma^2}{2} \Big ) t + \sigma W_t  \bigg ], \spc \spc t \geq 0,
\]
where $r, \sigma > 0$ represent interest rate and volatility, respectively, and $x > 0$ is the initial value. \\
The process $X$ represents the price of some asset, say, for example, a stock. \\
We are also given a payoff function $\phi: \R_+ \to \R_+$ s.t. the payoff of exercise of the derivatives contract we are concerned about is $\phi(X_t)$ at time $t$. \\
In the case of a put option with strike price of $K > 0$, we have $\phi(x) = (K - x)^+$. \\
Using Itô's lemma we find that $X$ satisfies the following SDE,
\[
    \df{X_t} = r X_t \df{t} + \sigma X_t \df{W_t}, \spc X_0 = x.
\]
Note that $X$ is strong Markov. \\

We introduce $V(x)$, the value function of the perpetual (no expiry) put option, defined as follows using the notation outlined in the above definition:
\[
    V(x) := \sup_{\tau \in \cS} \E^x \Big [ e^{-r\tau} (K - X_\tau)^+ \Big ],
\]
i.e., $V(x)$ is the value of the Snell envelope (see Section II.II The Snell Envelope) of $e^{-tr} (K - X_t)^+$ at time 0, given that $X$ starts at $x$. \\
Our goal for this chapter is to characterize $V$ and also the optimal stopping time that achieves the value $V(x)$. \\

\phantomsection\subsection*{\quad III.II Characterization of the Solution}\label{sec:III.II}
Let us start with a lemma that will help us find $\bL_X$.
\begin{lemma} $\text{[PR]}$
    Consider a process $C_t$ satisfying the SDE,
    \[
        \df{C_t} = \mu(C_t) \df{t} + \sigma(C_t) \df{B_t},
    \]
    where $B_t$ is a Brownian motion and $\mu, \sigma$ are both Lipschitz continuous. \\
    We then have $\bL_C = \mu(x) \frac{\partial}{\partial x} + \frac{1}{2} \big ( \sigma(x) \big )^2 \frac{\partial^2}{\partial x^2}$ (where we restrict the domain to $f \in C^2(\R)$ with $f'$ and $f''$ bounded).
\end{lemma}

We now direct our attention back to the problem at hand. \\
Applying Lemma 12 to $X$, we find $\bL_X = rx \frac{\partial}{\partial x} + \frac{\sigma^2}{2} x^2 \frac{\partial^2}{\partial x^2}$. \\
\\
Recall our discussion in the Markov Motivation section. \\
We use the intuition from there along with the fact that as $x$ decreases, it seems that we stand to gain less and less from waiting to exercise. \\
For now, we assume that the optimal stopping time is a hitting time of the form $\tau_b := \{t \quantifyspace X_t \leq b\}$ for some $0 < b < K$; we will not prove this yet, but rather use this as a starting place from which to find a candidate characterization of $V$. \\
We now list some properties that we expect $V$ to have and some reasoning why we expect this (again, we do not prove them; they just help us arrive at a candidate). \\
\begin{enumerate}
    \item[(i)]
If $\tau_b$ is optimal, we expect that the Snell envelope of $(e^{-tr} (K - X_{t})^+)_{t \geq 0}$ stopped at $t \wedge \tau_b$ is a martingale and so,
\[
    \bL_X V = r V, \spc  x > b.
\]
    \item [(ii)]
Due to optimality of $\tau_b$
\[
    V(x) = (K - x)^+, \spc x \leq b.
\]
    \item[(iii)]
Due to sub-optimality of exercise above $b$,
\[
    V(x) > (K - x)^+, \spc x > b.
\]
    \item[(iv)]
Necessitated for $V$ to be $C^1$ (we want $C^1$ so that we can apply Itô's lemma or Itô-Tanaka-Meyer, this is often called the \textbf{smooth fit principle}), and given we expect (ii) to hold,
\[
    V'(x) = -1, \spc x \leq b.
\]
\end{enumerate}

The most interesting of these is (i): suddenly we have an ODE. \\
Specifically we expect $V$ to satisfy,
\[
    rx V'(x) + \frac{\sigma^2}{2} x^2 V''(x) - r V(x) = 0 \spc \spc \forall x > b.
\]
Seeing that we have a sum of a $V$ term, an $x V'$ term, and an $x^2 V''$ term, it makes sense to look for a polynomial solution where all elements of the same power cancel. \\
We consider a solution of the form $x^p$; substituting this candidate, we need $p$ to satisfy,
\[
    x^p \Big ( rp + \frac{\sigma^2}{2} p [p-1] - r \Big ) = 0 \spc \spc \forall x > b.
\]
Specifically, either $p = 1$ or $p = \frac{- 2 r}{\sigma^2}$. \\
Intuitively, we expect $\lim_{x \to \infty} V(x) = 0$, so having a $p > 0$ term does not make much sense; this leaves us with $p = -\gamma := \frac{-2 r}{\sigma^2}$. \\
And so we introduce the candidate,
\[
    H (x) := \begin{cases}
        C x^{-\gamma} & x > b \\
        K - x                & x \leq b < K
    \end{cases}.
\]
Now our task becomes finding suitable values of $b$ and $C$. \\
We need the candidate to be continuous at $b$. Also we want to be able to apply Itô's lemma or Itô-Tanaka-Meyer, and so the smooth fit principle tells us that we want $H'$ to be continuous at $b$. \\
%We need the candidate and its derivative to be continuous at $b$, 
Now we get the following system:
\[
    H(b) = K - b = C b^{-\gamma},
\]
\[
    H'(b) = -1 = - \gamma C b^{-\gamma - 1}.
\]
We first solve both for $C$ in terms of $b$ and then take these two expressions and solve for $b$ to find $b = \frac{K}{1 + \gamma^{-1}}$. \\
Then we find $C = \Big ( \frac{K}{1 + \gamma^{-1}} \Big )^{\gamma + 1} \gamma^{-1}$. \\
Now that we have our candidate, our task is to verify it. \\
\begin{thm}
    The candidate $H(x)$ is equal to the value function $V(x)$, i.e.
    $V(x) = H(x)$ for all $x \in (0,\infty)$.

    \proof
    We define $\daleth_t := e^{-rt}$ and $\beth_t := H(X_t)$.
    From simple calculus, we know $\df{\daleth_t} = -r \daleth_t \df{t}$. \\
    Since $H'$ is absolutely continuous, we can use Itô-Tanaka-Meyer to find,
    \[
        \df{\beth_t} = H'(X_t) \df{X_t} + \frac{1}{2} H''(X_t) \df{\la X, X \ra_t}.
    \]
    Recall from Itô-Tanaka-Meyer that this is well-defined even though $H''$ is not defined everywhere.
    Since $\daleth_t$ is of finite variation, we get that $\la \daleth, \beth \ra_t = 0$. \\
    Now, using 2-dimensional Itô's lemma on $(a,b) \mapsto ab$, we find,
    \[
        \df{(\daleth_t \beth_t)} = \daleth_t \df{\beth_t} + \beth_t \df{\daleth_t}.
    \]
    Now,
    \[
        \df{(\daleth_t \beth_t)} = \daleth_t (H'(X_t) \df{X_t} + \frac{1}{2} H''(X_t) \df{\la X, X \ra_t}) + \beth_t (-r \daleth_t \df{t}).
    \]
    Recall $\df{X_t}$ and the definitions of $\daleth_t$ and $\beth_t$. \\
    By grouping $\df{u}$ terms together and integrating we find,
    \[
        e^{-rt} H(X_t) = H(X_0) + \int_0^t e^{-ru } (\bL_X H - r H)(X_u) \df{u} +
    \]
    \[
        \int_0^t e^{-ru} \sigma X_u H'(X_u) \df{W_u} \spc \spc \forall t \geq 0 \spc \als
    \]
    First, note that in the domain $(b, \infty)$, we have $(\bL_X H - r H)(x) = 0$ from our construction of $V$. \\
    In the $(0,b)$ domain, we have $(\bL_X H - rH)(x) = -rx - r(K - x) = -rK < 0$ and so we have $(\bL_X H - rH)(x) \leq 0$ where it is defined. \\
    \\
    Now we define $M_t := \int_0^t e^{-ru} \sigma X_u H'(X_u) \df{W_u}$. \\
    Since $W$ is a martingale and the integrand is adapted and continuous, it is progressively measurable and in $L^2_{loc}(W)$. Thus, from Theorem 6, we have that $M_t$ is a local martingale starting at 0. \\
    We select a localizing sequence of $M_t$, say $\{\tau_n\}_{n=1}^\infty$. \\
    \\
    Due to the integrand in the $du$ integral being negative we have,
    \[
        e^{-rt} H(X_t) \leq H(X_0) + M_t.
    \]
    Combining properties (ii) and (iii) of our construction of $H$, we see,
    \[
        (K - x)^+ \leq H(x) \spc \spc \forall x > 0,
    \]
    and so,
    \[
        e^{-r s} (K - X_s)^+ \leq e^{-rs} H(X_s) \leq H(X_0) + M_s.
    \]
    Now fix a stopping time $\tau$ and consider
    \[
        e^{-r \tau \wedge \tau_n} (K - X_{\tau \wedge \tau_n}) \leq H(X_0) + M_{\tau \wedge \tau_n}.
    \]
    Fix some $x > 0$ to be the starting point of our process. \\
    Take expectation $\E^x$ of both sides to see,
    \[
        \E^x \big [
            e^{-r \tau \wedge \tau_n} (K - X_{\tau \wedge \tau_n})
        \big ] \leq H(x).
    \]
    Note that the above holds for all $n$. Now apply Fatou's lemma:
    \[
        \E^x[e^{-r \tau} (K - X_{\tau})^+] \leq H(x).
    \]
    This shows that $V(x) \leq H(x)$. \\
    \\
    Now we need to show the reverse. \\
    Fix some $x > 0$; in the case that $x \leq b$ then $H(x) = K - x \leq V(x)$, so we consider the case that $x > b$. \\
    Note that since $\bL_X H - r H = 0$ on $(b, \infty)$, we have that for all $n$,
    \[
        \Big ( e^{ - r (t \wedge \tau_b \wedge \tau_n)} H(X_{t \wedge \tau_b \wedge \tau_n}) = H(x) + M_{t \wedge \tau_b \wedge \tau_n} \Big )_{t \geq 0} \text{ is a martingale w.r.t. } \Prb^x.
    \]
    And so,
    \[
        H(x) = \E^x [e^{-r \tau_b \wedge \tau_n} H(X_{\tau_b \wedge \tau_n})]. % = \E^x[e^{-r (\tau_b \wedge \tau_n)} (K - X_{\tau_b \wedge \tau_n})] \leq \sup_\tau \E^x [e^{-r \tau} (K - X_\tau)] = V(x)
    \]
    Since $e^{-r \tau_b \wedge \tau_n} H(X_{\tau_b \wedge \tau_n})$ for all $n$ is dominated by the constant $K$, we can apply the dominated convergence theorem to find,
    \[
        H(x) = \E^x [e^{-r \tau_b} H(X_{\tau_b})] = \E^x[e^{-r \tau_b} H(X_{\tau_b}) \ind{\tau_b < \infty}] + \E[e^{-r \tau_b} H(X_{\tau_b}) \ind{\tau_b = \infty}] =
    \]
    \[
        \E^x[e^{-r \tau_b} (K - b)^+ \ind{\tau_b < \infty}] =
        \E^x[e^{-r \tau_b} (K - X_{\tau_b})^+] \leq
    \]
    \[
        \sup_\tau \E^x [e^{-r \tau} (K - X_\tau)^+] = V(x).
    \]
    We have now shown $H(x) = V(x)$ for all $x > 0$ and that $\tau_b$ achieves $V(x)$. \\
    \QED
\end{thm}

Now we want to see that the Snell envelope is exactly that which our Markov intuition would suggest.

\begin{thm}
    The Snell envelope of $e^{-rt} (K - X_t)^+$ is $e^{-rt} V(X_t)$.
    \proof
We recall that in the proof of the previous theorem we found a decomposition of $e^{-rt} V(X_t) = V(X_0) + M_t - A_t$, where $M_t$ is a local martingale and $A_t$ is a non-decreasing process. \\
Fix some $\tau \in \cS$ and some $\theta \in \cS_\tau$. \\
Now we have the following,
\[
    e^{-r\theta}(K - X_\theta)^+ \leq e^{-r\theta} V(X_\theta) = e^{-r\tau} V(X_\tau) + (M_\theta - M_\tau) - (A_\theta - A_\tau) \leq
\]
\[
    e^{-r\tau} V(X_\tau) + (M_\theta - M_\tau) \spc \als.
\]
Now take the minimum of these times stopped at $\tau_n$, where $\{\tau_n\}_{n=1}^\infty$ is a localizing sequence of $M_t$:
\[
    e^{-r (\theta \wedge \tau_n)} (K - X_{\theta \wedge \tau_n})^+ \leq e^{-r(\tau \wedge \tau_n)} V(X_{\tau  \wedge \tau_n}) + M_{\theta \wedge \tau_n} - M_{\tau \wedge \tau_n}.
\]
Now take conditional expectation w.r.t. $\cF_\tau$ and, because $M^{\tau_n}$ is a martingale, we can move terms around to get
\[
    \E \big [ e^{-r (\theta \wedge \tau_n)} (K - X_{\theta \wedge \tau_n})^+ - e^{-r (\tau \wedge \tau_n)} V(X_{\tau \wedge \tau_n}) \quantifyspace \cF_\tau \big ] \leq 0.
\]
Now, due to the fact that the inside of the conditional expectation above is dominated by $2K$, we can use dominated convergence for conditional expectation to get,
\[
    \E[e^{-r \theta}(K - X_\theta) \quantifyspace \cF_\tau] \leq e^{-r \tau}V(X_\tau) \spc \als.
\]
This demonstrates that $\esssup_{\theta \in \cS_\tau} \E[e^{-r \theta} (K - X_\theta)] \leq e^{-r \tau} V(X_\tau)$ a.s.. \\
\\
Now we want to show the reverse. \\
We can decompose $F_t := e^{-rt} V(X_{t}) = V(x) + M_t - A_t$. \\
We define $\rho := \inf \{t \quantifyspace t \geq \tau, \sspc X_t \leq b \}$. \\
We note that $A_t = \int_0^t e^{-ru} (\bL_X H - r H)(X_u) \df{u}$ is constant on the interval $[\tau, \rho]$. \\
\\
We consider $\{\tau_n\}_{n=1}^\infty$ to be a localizing sequence of $(M^{\rho}_t)_{t \geq 0}$. \\
Now we fix some $\tau \in \cS$ and some $n \in \N$. \\
From properties of conditional expectation we find,
\[
    F_{\tau \wedge \tau_n} = \E[F_{\rho \wedge \tau_n} \quantifyspace \cF_{\tau \wedge \tau_n}] =
    \E[F_{\rho \wedge \tau_n} \quantifyspace \cF_\tau] \spc \als.
\]
Again we can apply conditional dominated convergence to get,
\[
    F_\tau = \E[F_\rho \quantifyspace \cF_\tau] = \E[e^{-r \rho} (K - X_\rho)^+ \quantifyspace \cF_\tau] \spc \als.
\]
This shows that $e^{-r \tau} V(X_\tau) \leq \esssup_{\theta \in \cS_\tau} \E[e^{-r \theta} (K - X_\theta)^+ \quantifyspace \cF_\tau]$ a.s.. \\
This proves the theorem and that $\rho$ is optimal for stopping after $\tau$. \\
\QED \\
\end{thm}

With the proof of this theorem, we have now achieved what our intuition in the Markov Motivation section informed us might be possible. 
We used our intuition to find an explicit characterization of a Snell Envelope candidate and then went about proving it is indeed the Snell Envelope. 
Most notably, we found that the Snell Envelope is a product of the imposed scale-invariant time penalty function $e^{-rt}$ and a value function which is dependent only on the current state.
We also saw that the optimal stopping time for stopping after some stopping time $\tau$ is a hitting time.
This matches nicely with our intuition coming from the fact that $e^{-rt} (K - X_t)$ is strong Markov, and that taking the positive portion of this is a deterministic function of a strong Markov process. 


\phantomsection\section*{Part IV: Ornstein-Uhlenbeck Optimal Stopping Example}\label{sec:part4}
In this chapter, we examine Example 10.1 from Chapter 10.2 of [PS], which is another example of the conversion of an optimal stopping problem to a free boundary problem. \\
We are given the stopping problem with value function as follows,
\[
    V_*(t, x) = \sup_{\tau \in S^x_X} \E^x \Big (
        |X_\tau| - c \sqrt{t + \tau}
    \Big ),
\]
where $(X_t)_{t \geq 0}$ is a Brownian motion plus a constant $x$ under the probability measure $\Prb^x$, and we take $\E^x$ to be integration w.r.t. $\Prb^x$, and $S^x_X$ is the set of stopping times $\tau$ w.r.t. $(\cF^X_t)_{t \geq 0}$ s.t. $\E^x [\sqrt{\tau}] < \infty$. \\
\\
We take $\sigma_t := e^{2t} - 1$; now we define $Z_t := X_{\sigma_t} / \sqrt{1 + \sigma_t}$, $t \geq 0$. \\
Using Itô's lemma and quadratic variation arguments, one can find
\[
    \df{Z_t} = - Z_t \df{t} + \sqrt{2} \df{B_t}, \spc Z_0 = x,
\]
where $B_t$ is a Brownian motion. \\
Now apply Lemma 12 to get,
\[
    \bL_Z = -z \frac{\partial}{\partial z} + \frac{\partial^2}{\partial z^2}.
\]
\begin{remark}
    The stopping problem in this section is no longer of a time-homogeneous structure;
    % note that $|X_\tau| - c \sqrt{t + \tau}$ is not strong Markov due to the structure of the time term; 
    in fact, it does not seem that this is a deterministic function of a time homogenous strong Markov process.
    This said, when we have some extra time dependence we can introduce time into a new state space, i.e., if we had state space $(E, \cE)$ and a time homogeneous strong Markov process $X_t$ on this space we can construct state space $(E \times \R_+, \cE \otimes \cB(\R_+))$ and in this new state space we can reasonably expect to have the time homogeneous strong Markov property for $(X_t, t)$.
    This should give us intuition that the optimal stopping time for such problems is a function only of state in the sense of $(E, \cE)$ and of time. \\
\end{remark}

\phantomsection\subsection*{IV.I Reduction of Value Function to 1 Variable}\label{sec:IV.I}
From the Paul Lévy characterization of Brownian motion, note that for a fixed $t > 0$, $W_s := \frac{1}{\sqrt{t}} B_{ts}$ is a Brownian motion. \\
Consider $\tau$ a stopping time w.r.t. the filtration generated by $B_t$; note that $\tau/t$ is a stopping time w.r.t. the filtration generated by $W_t$, and vice versa, by multiplying stopping times by $t$. \\
We define $S^x_B$ and $S^x_W$ analogously to $S^x_X$. \\
And so,
\[
    V_*(t, x) = \sup_{\tau \in S^x_B} \E \Big [ |B_\tau + x| - c \sqrt{t + \tau} \Big ] =
\]
\[
    \sqrt{t} \sup_{\tau/t \in S^x_W} \E \bigg [ \Big | \frac{1}{\sqrt{t}} B_{t (\tau/t)} + \frac{x}{\sqrt{t}} \Big | - c \sqrt{1 + \tau/t} \bigg ].
\]
In this manner, we have reduced our value function to a single variable,
\[
    V_*(t, x) = \sqrt{t} V_*(1, x/\sqrt{t}).
\]
\phantomsection\subsection*{IV.II Conversion to Ornstein-Uhlenbeck Magnitude Stopping}\label{sec:IV.II}
Observe the following identity,
\[
    |X_{\sigma_\tau}| - c \sqrt{1 + \sigma_\tau} = \sqrt{1 + \sigma_\tau} (|Z_\tau| - c) = e^\tau (|Z_\tau| - c).
\]
Due to the manner in which $Z_t$ is constructed as a function of $X_{\sigma_t}$, we have that $\cF^Z_t = \cF^X_{\sigma_t}$, so $\tau$ is a stopping time w.r.t. the filtration generated by $Z$ iff $\sigma_\tau$ is a stopping time w.r.t. the filtration generated by $X$. \\
Note that $\E^x[\sqrt{\sigma_\tau}] < \infty$ if and only if $\E^x[ e^\tau ] < \infty$. \\
We take $T^x$ to be the set of stopping times $\tau$ w.r.t. $\cF^Z_t$ s.t. $\E^x[e^\tau] < \infty$. \\
This correspondence between stopping times gives us,
\[
    W_*(x) := V_*(1, x) =
    % \sup_{\sigma_\tau \in S^x_{X_\sigma}} \E^x [|X_{\sigma_\tau}| - c \sqrt{1 + \sigma_\tau} ] =
    \sup_{\tau \in T^x} \E^x [e^\tau (|Z_\tau |- c) ].
\]
For the sake of clarity, we note that $Z_0 = X_0$ and so, if we ever refer to $W_*(z)$ or write $\E^z$, these have the same meaning as we have defined already, and there should be no confusion from the semantics. \\

\phantomsection\subsection*{IV.III Conversion to Free Boundary Problem}\label{sec:IV.III}
With $Z_t$ being an Ornstein-Uhlenbeck process, we note that $e^t (|Z_t| - c)$ does not seem to be a deterministic function of a strong Markov process but $(|Z_t| - c)$ is strong Markov and the $e^t$ term maintains a scale invariance. Heuristically, if state-time is $(z, t)$ and this is in the continuation region because we can stop in a manner that improves expected reward by, say, $k$\% over stopping immediately, then at $(z, t+h)$ we expect that we can again stop in a manner that improves expected reward by $k$\% over stopping immediately. That is, our intuition suggests that the time induced change in transition dynamics should not introduce time dependence into the optimal stopping policy. \\
\\
We assume that the optimal stopping time is of the form $\tau_{z_*} := \inf \{t \geq 0 \quantifyspace |Z_t| \geq z_*\}$, where $z_* > 0$ is a constant. \\
We can now set up the free boundary problem. \\
We search for a candidate solution $W(z)$ that satisfies the following,
\begin{enumerate}
    \item[(i)] Since we expect $e^t W(Z_t)$ to be a martingale in $z \in (-z_*, z_*)$:
    \[
        \bL_Z W(z) = - W(z).
    \]
    \item[(ii)] Due to optimality of stopping at $\tau_{z_*}$:
    \[
        W(\pm z_*) = z_* - c.
    \]
    \item[(iii)] Due to option to stop immediately:
    \[
        W(z) \geq |z| - c \spc \spc \forall z.
    \]
    \item[(iv)] To maintain $W \in C^1$ so that we can hope to apply Itô's lemma or Itô-Tanaka-Meyer (smooth fit principle) we expect:
    \[
        W'(\pm z_*) = \pm 1.
    \]
    \item[(v)] Correct sign of slope, sign$: \R \to \{1, -1, 0\}$:
    \[
        \text{sign } W'(z) = \text{ sign } z \spc \spc \forall z.
    \]
\end{enumerate}

\phantomsection\subsection*{IV.IV Deriving the Candidate}\label{sec:IV.IV}
Along the lines of chapter 10 of [PS], we now introduce a function which will be helpful for characterizing the solution in the continuation region.

\begin{defn}
    The \textbf{Kummer confluent hypergeometric} function is given by,
    \[
        M(a, b, x) = \sum_{k=0}^\infty \frac{(a)_k \sspc x^k}{(b)_k \sspc k!}, \spc \spc a,b,x \in \C,
    \]
    where $(\rho)_k$ represents the product $\rho (\rho + 1) .. (\rho + k - 1)$. Using standard analysis techniques, this function can be seen to be well-defined and in $C^2$.
\end{defn}

Importantly, note that $z \mapsto M(\frac{-1}{2}, \frac{1}{2}, \frac{z^2}{2})$ and $z \mapsto z$ are linearly independent and solve $\bL_Z W(z) = -W(z)$. \\
By Corollary 4.35 from [TA], the solution space for a second-order ODE is 2-dimensional, and so the two aforementioned functions form a basis of the solution space to $\bL_Z W(z) = -W$. \\
In fact, since $-Z_t$ follows the same SDE as $Z_t$ (with the corresponding negative Brownian motion), we expect $W$ to be an even function. \\
Since the first of our basis functions is even and the second is odd, we expect our solution to be of the form,
\[
    W(z) = \begin{cases}
        - A M(\frac{-1}{2}, \frac{1}{2}, \frac{z^2}{2}) & z \in (-z_*, z_*) \\
        |z| - c & z \not \in (-z_*, z_*)
    \end{cases},
\]
where $z_* > 0$ and $A \in \R$ are constants and the hypothesized optimal stopping time is $\tau_{z_*} := \inf \{t \quantifyspace |Z_t| \geq z_* \}$. \\
\\
Note that $M(\frac{-1}{2}, \frac{1}{2}, 0) = 1$ and $\lim_{z \to \infty} M(\frac{-1}{2}, \frac{1}{2}, \frac{z^2}{2}) = - \infty$ and $M(\frac{-1}{2}, \frac{1}{2}, \frac{z^2}{2})$ is strictly decreasing in $z$ on $\R_+$, which is easily seen from the non-constant terms in the summation having negative coefficients, and so must have a unique positive root. \\
Define $r$ to be the unique positive root of $M(\frac{-1}{2}, \frac{1}{2}, \frac{z^2}{2})$. \\
From conditions (ii) and (v), we find $c \geq r$. This tells us that existence of a solution meeting the criteria of $W$ is incompatible with $c < r$; this may be a hint that it is never optimal to stop in the case that $c < r$, giving $W_*(z) = \infty$. \\
We now assume that $c > r$ (we will handle $c = r$ later). \\
\\
In order to maintain continuity, we must select $- A = \frac{z_* - c}{M \big (\frac{-1}{2}, \frac{1}{2}, \frac{z_*^2}{2} \big )}$. \\
Also note due to the smooth fit principle we expect $W \in C^1$; we expect to have that $W'(\pm z_*) = \pm 1$, and this gives us another relation for $A$. \\
Using $\frac{d}{\df{x}} M(a,b,x) = \frac{a}{b} M(a + 1, b+1, x)$, we find,
\[
    A = \Big ( z_* M \Big (\frac{1}{2}, \frac{3}{2}, \frac{z_*^2}{2} \Big ) \Big )^{-1} = \frac{-(z_* - c)}{M \big (\frac{-1}{2}, \frac{1}{2}, \frac{z_*^2}{2} \big )}.
\]
Rearranging gives,
\[
    (c - z_*) z_* = \frac{M \big (\frac{-1}{2}, \frac{1}{2}, \frac{z^2_*}{2} \big )}{M \big (\frac{1}{2},\frac{3}{2},\frac{z_*^2}{2} \big )}.
\]
From the intermediate value theorem, we find that there must be some $z_* \in (0, r)$ satisfying the above.
Rearranging again, we find that such a $z_*$ is a zero of,
\[
    q(z) = (c - z)z M \Big ( \frac{1}{2},\frac{3}{2},\frac{z^2}{2} \Big ) - M \Big (\frac{-1}{2},\frac{1}{2},\frac{z^2}{2} \Big ).
\]
Taking the derivative of the above, we find,
\[
    q'(z) = \Big ( \frac{(c - z)z^2}{3} \Big ) M \Big (\frac{3}{2},\frac{5}{2},\frac{z^2}{2} \Big ) + (c - z) M \Big ( \frac{1}{2},\frac{3}{2},\frac{z^2}{2} \Big ).
\]
Note that $q'$ is strictly positive on $z \in (0, c) \supset (0, r)$, and so a positive root $z_* \in (0, r)$ exists and is unique. We will now let $z_*$ denote this root. \\

\phantomsection\subsection*{IV.V Verify the Candidate}\label{sec:IV.V}
We now have our candidate satisfying criteria (i)-(v) and our task is to prove that $W(z) = W_*(z)$. \\
\\
We now define $F_t := e^t W(Z_t)$. \\

First, we can apply Itô-Tanaka-Meyer to find $\df{(W(Z_t))}$ (since $W$ is $C^1$ with absolutely continuous derivative) and calculus to find $\df{e^t}$, and then apply Itô's lemma to the product map $(a, b) \mapsto ab$ to find,
\[
    F_t = F_0 + \int_0^t F_u \df{u} + \int_0^t e^u W'(Z_u) \df{Z_u} + \frac{1}{2} \int_0^t 2 e^u W''(Z_u) \df{u} \spc \forall t \sspc \als.
\]
Recalling the SDE that $Z_t$ satisfies and rearranging terms, we find,
\[
    F_t = F_0 + \int_0^t e^u (\bL_Z W + W)(Z_u) \df{u} + \sqrt{2} \int_0^t e^u W'(Z_u) \df{B_u}.
\]
We decompose as expected,
\[
    A_t := - \int_0^t e^u (\bL_Z W + W)(Z_u) \df{u} \spc \text{ and } \spc M_t := \sqrt{2} \int_0^t e^u W'(Z_u) \df{B_u},
\]
and so we get,
\[
    F_t = F_0 - A_t + M_t.
\]
Note that $\bL_Z W + W \leq 0$ and so $A_t$ is non-decreasing.
Also, from Theorem 6, we get that $M_t$ is a local martingale starting at 0; let $\{\tau_n\}_{n=1}^\infty$ be a localizing sequence of bounded stopping times. \\
\\
For the sake of convenience we define $G_t := e^t (|Z_t| - c) \leq e^t W(Z_t) = F_t$. \\
Fix some stopping time $\tau$ s.t. $\E^z[e^\tau] < \infty$ and some $n \in \N$:
\[
    G_{\tau \wedge \tau_n} \leq F_{\tau \wedge \tau_n} = F_0 - A_{\tau \wedge \tau_n} + M_{\tau \wedge \tau_n} \leq F_0 + M_{\tau \wedge \tau_n}.
\]
Take expectation $\E^z$ of both sides:
\[
    \E^z[G_{\tau \wedge \tau_n}] \leq \E^z[F_0] = W(z).
\]
Because $\tau$ is chosen s.t. $\E^z[e^\tau] < \infty$, $-c e^\tau$ is an integrable lower bound of $G_{\tau \wedge \tau_n}$, by Fatou's lemma,
\[
    \E^z [G_\tau] \leq W(z).
\]
This shows that $W_*(z) \leq W(z)$. \\
\\
Now we want to show the reverse. \\
Recall that for each $n$, we have that $(F^{\tau_n \wedge \tau_{z_*}}_t)_{t \geq 0}$ is a martingale. \\
Also, we know from our construction of $W$ that $F_{\tau_{z_*}} = G_{\tau_{z_*}}$ on $\{\tau_{z_*} < \infty\}$, which has probability 1 under each $\Prb^z$. \\
Because $(z_* - c) \leq 0$ and $W(z) < 0$ for $|z| \leq z_* < r$, for the non-trivial case of $|z| < z_*$,
\[
    W(z) = \E^z[F_0] = \E^z [F_{\tau_n \wedge \tau_{z_*}}] = \E^z [ \ind{\tau_n > \tau_{z_*}} G_{\tau_{z_*}} ] + \E^z[\ind{\tau_n \leq \tau_{z_*}} F_{\tau_n}] \leq
\]
\[
    \E^z [ \ind{\tau_n > \tau_{z_*}} G_{\tau_{z_*}} ].
\]
Since the random variable $G_{\tau_{z_*}}$ is non-positive, we can take the negative of the above inequality, use monotone convergence, and take the negative of the new inequality to get,
\[
    W(z) \leq \E^z [G_{\tau_{z_*}}].
\]
If we can show that $\E[e^{\tau_{z_*}}] < \infty$, we will have shown $W(z) = W_*(z)$. \\
\\
Define $f(z) := \frac{M \big (\frac{-1}{2},\frac{1}{2},\frac{z^2}{2} \big )}{M \big (\frac{-1}{2},\frac{1}{2},\frac{z_*^2}{2} \big )} \geq 0$. \\
Note that because $z_* \in (0, r)$, we have that the numerator and denominator of the above have the same sign for $z \in [0, z_*]$ and so $f(z)$ is non-negative for $z \in [0, z_*]$. \\
\\
Because $\bL_Z f = -f$, using Itô's lemma, we find that $Y_t := e^t f(Z_t)$ is a local martingale. \\
Also note,
\[
    |Y_t^{\tau_{z_*} \wedge n}| \leq e^n \max f([-z_*, z_*]) =: C_n < \infty.
\]
And so, by Lemma 2, we have that $(Y^{\tau_{z_*} \wedge n}_t)_{t \geq 0}$ is a true martingale. \\
Now,
\[
    \E^z[Y_{n \wedge \tau_{z_*}}] = \E^z[Y_0] = f(z).
\]
Because $Z_t$ is an Ornstein-Uhlenbeck process, $\tau_{z_*} < \infty$ a.s.. \\
We use that $Y_t$ is non-negative to apply Fatou's lemma and find,
\[
    \E^z[e^{\tau_{z_*}}] = f(z_*) \E^z[ e^{\tau_{z_*}}] = \E^z[Y_{\tau_{z_*}}] \leq f(z).
\]
And so we have shown that in the case $c > r$, $W(z) = W_*(z)$. \\

From the $c > r$ case, we can also get the $c = r$ case. \\
We let $W_c$ and $W_{c, *}$ represent $W$ and $W_*$ as before except now with $c$ specified explicitly. \\
We take $\{c_n\}_{n=1}^\infty$ a decreasing sequence converging to $r$ and fix some $z$. \\
Due to continuity of $c \mapsto W_{c}(z)$ and the fact that all $e^\tau(|Z_\tau| - c_n)$ have a common integrable lower bound $- c_1 e^\tau$, for $c \geq r$ we can use monotone convergence to find,
\[
    W_{r, *}(z) = \sup_{\tau \in T^z} \E^z[e^\tau (|Z_\tau| - r)] =
    \sup_{\tau \in T^z} \bigg ( \sup_{n \in \N} \E^z[e^\tau (|Z_\tau| - c_n)]  \bigg ) =
\]
\[
    \sup_{n \in \N} \bigg ( \sup_{\tau \in T^z} \E^z[e^\tau (|Z_\tau| - c_n)]  \bigg ) =
    \sup_{n \in \N} W_{c_n, *}(z) =
    \sup_{n \in \N} W_{c_n}(z) = W_{r}(z).
\]
\cheekynewline
Putting it all together, we have now shown,
\begin{thm} $\text{[PS]}$
    Given $c \geq r$,
    \[
        V(t, x) = \sup_\tau \E^x  \Big ( |X_\tau| - c \sqrt{t + \tau} \Big ) =
    \]
    \[
        \begin{cases}
            - \sqrt{t} z_*^{-1} M \Big (-\frac{1}{2}, \frac{1}{2}, \frac{x^2}{2t} \Big) \Big [ M \Big (\frac{1}{2}, \frac{3}{2}, \frac{z_*^2}{2} \Big) \Big ]^{-1} & \frac{|x|}{\sqrt{t}} < z_* \\
            |x| - c \sqrt{t} & \frac{|x|}{\sqrt{t}} \geq z_*
        \end{cases}.
    \]
    The optimal stopping time for the original problem is given by:
    \[
        \tau_* = \inf \{s > 0 \quantifyspace |X_s| \geq z_* \sqrt{t + s}\}.
    \]
    For $c = r$, the stopping times $\tau_*$ from Theorem 12 are approximately optimal if we let $c \downarrow r$. \\
\end{thm}
\cheekynewline
We also have the following theorem for the $c < r$ case. \\


\begin{thm}
    For $c < r$ we have $V_*(t, x) = \infty$ and it is never optimal to stop.
    \proof
 
    Take $\alpha := \frac{c + r}{2} \in (c, r)$. \\
    We have already seen, en route to the proof of the previous theorem, that
    \[
        \E^z[e^\tau] < \infty \spc \forall z.
    \]
    Where $\tau := \inf \{t \geq 0 \quantifyspace |Z_t| \geq \alpha\}$. \\
    Similarly, for $\rho_t := \inf \{s \geq t \quantifyspace |Z_s| \geq \alpha \}$ we have $\E^z [e^{\rho_t}] < \infty $ for all $z$. \\
    % In the proof of Theorem 12 we saw that $\E^z[e^{\rho_0}] < \infty$.
    % In a similar manner to $\rho_0$ we see that $\E^z[e^{\rho_t}] < \infty$.
    Now for any fixed $t$,
    \[
        \Prb^z[\{\rho_t = \infty\}] = 0 \spc \spc \forall z.
    \]
    This implies,
    \[
        (\alpha - c) e^t \leq (\alpha - c) e^{\rho_t} = (|Z_{\rho_t}| - c)e^{\rho_t} \spc \spc \Prb^z\text{-}\als.
    \]
    And so,
    \[
        W_{c, *}(z) \geq (\alpha - c) e^t.
    \]
    Since this holds for all $t$, we have that $W_{c,*}(z) = \infty$ for all $z$. \\
    \QED
\end{thm}
\cheekynewline
In this most recent example, we have again used our Markov intuition to find suitable conditions that a reasonable candidate value function must satisfy.
Using these conditions, we then found an explicit characterization for the candidate value function and used other tools to verify it is indeed correct. \\
